# Public Subnets
resource "aws_subnet" "public" {
  for_each = {
    for i in range(3) :
    "subnet_${i + 1}" => {
      cidr_block        = cidrsubnet(aws_vpc.main.cidr_block, 4, i)
      availability_zone = data.aws_availability_zones.available.names[i]
      name              = "public-subnet-${i + 1}"
    }
  }
  vpc_id             = aws_vpc.main.id
  cidr_block        = each.value.cidr_block
  availability_zone = each.value.availability_zone
  map_public_ip_on_launch = false
  tags = {
    Name = each.value.name
  }
}

# NAT Gateways
resource "aws_nat_gateway" "main" {
  for_each = {
    for i in range(3) :
    "nat_gateway_${i + 1}" => {
      eip_id    = aws_eip.nat["nat_eip_${i + 1}"].id
      subnet_id = aws_subnet.public["subnet_${i + 1}"].id
      name      = "nat-gateway-${i + 1}"
    }
  }
  allocation_id = each.value.eip_id
  subnet_id     = each.value.subnet_id
  tags = {
    Name = each.value.name
  }
  depends_on = [aws_internet_gateway.main]
}
