# Amazon Managed Domain Lists and Associate with Egress Rule
resource "aws _route53_ resolver_firewall_rule" "block_amazon _managed _domains" {
    count = length(var managed_domain_list_ids)
    firewall_rule_group_id = aws_route53_resolver_firewall rule_group.egress.id
    priority= 100 + count.index # Ensure each rule has a unique priority
    action= "ALERT"


    name = "block-amazon-managed-domain-${count. index}"

    firewall domain_list_id - var.managed_domain_list_ids[count.index]
}
